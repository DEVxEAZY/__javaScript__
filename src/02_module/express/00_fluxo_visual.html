// equivalente em código — ordem dos use() importa server.js 
const express = require("express");
const session = require("express-session");
const MongoStore = require("connect-mongo");
const helmet  = require("helmet");
const flash   = require("connect-flash");
const csrf    = require("csurf");

const app = express();

// (01 → 07) middlewares na ordem em que rodam
app.use(helmet());
app.use(express.static("public"));
app.use(express.urlencoded({ extended: true }));
app.use(express.json());
app.use(session({ store: MongoStore.create({ mongoUrl }), secret, resave: false }));
app.use(flash());
app.use(csrf());
app.use(meusMiddlewaresCustom);

// (08 → 11) rotas → controllers → models → views
app.set("view engine", "ejs");
app.use("/produtos", produtoRouter);

// (!!) error handler com 4 args, por último
app.use((err, req, res, next) => res.status(500).render("erro", { err }));

app.listen(3000);
